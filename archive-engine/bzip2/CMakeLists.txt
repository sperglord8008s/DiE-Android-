cmake_minimum_required(VERSION 3.16)
project(die_archive_bzip2 C)
include(GNUInstallDirs)
if(NOT EXISTS "${BZIP2_SOURCE_ROOT}/bzlib.h")
  message(FATAL_ERROR "Set BZIP2_SOURCE_ROOT to the verified original source tree")
endif()
set(sources blocksort.c huffman.c crctable.c randtable.c compress.c decompress.c bzlib.c)
list(TRANSFORM sources PREPEND "${BZIP2_SOURCE_ROOT}/")
add_library(bz2 STATIC ${sources})
set_target_properties(bz2 PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(bz2 PUBLIC "${BZIP2_SOURCE_ROOT}")
install(TARGETS bz2 ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${BZIP2_SOURCE_ROOT}/bzlib.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
