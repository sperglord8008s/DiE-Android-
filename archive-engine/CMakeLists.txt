cmake_minimum_required(VERSION 3.16)
project(diearchive C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  set(archive_mode android)
else()
  set(archive_mode host)
endif()
set(archive_build "${CMAKE_CURRENT_SOURCE_DIR}/../app/build/archive-libraries-${archive_mode}")
set(archive_prefix "${archive_build}/prefix")
set(archive_source "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/Archive/libarchive/libarchive")
foreach(codec bz2 lzma lz4 zstd)
  add_library(archive_${codec} STATIC IMPORTED)
  set_target_properties(archive_${codec} PROPERTIES IMPORTED_LOCATION "${archive_prefix}/lib/lib${codec}.a")
endforeach()
add_library(archive_upstream STATIC IMPORTED)
set_target_properties(archive_upstream PROPERTIES IMPORTED_LOCATION "${archive_build}/libarchive/libarchive/libarchive.a")
find_package(ZLIB REQUIRED)
add_library(diearchive SHARED archive_reader.cpp probe.cpp decoder_limits.c)
set_target_properties(diearchive PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(diearchive PRIVATE "${archive_source}" "${archive_prefix}/include")
target_compile_features(diearchive PRIVATE cxx_std_17)
target_compile_definitions(diearchive PRIVATE _FILE_OFFSET_BITS=64)
target_compile_options(diearchive PRIVATE -O2 -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_link_libraries(diearchive PRIVATE archive_upstream archive_bz2 archive_lzma archive_lz4 archive_zstd ZLIB::ZLIB)
target_link_options(diearchive PRIVATE -Wl,--no-undefined -Wl,--gc-sections "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
foreach(decoder lzma_stream_decoder lzma_auto_decoder lzma_alone_decoder lzma_lzip_decoder lzma_raw_decoder ZSTD_createDStream)
  target_link_options(diearchive PRIVATE "-Wl,--wrap=${decoder}")
endforeach()
set_property(TARGET diearchive APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
if(ANDROID)
  target_link_options(diearchive PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
  find_package(JNI REQUIRED)
  target_include_directories(diearchive PRIVATE ${JNI_INCLUDE_DIRS})
endif()
