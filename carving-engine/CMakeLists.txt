cmake_minimum_required(VERSION 3.16)
project(diephotorec C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PHOTOREC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/PhotoRec")
include("${PHOTOREC_ROOT}/diemobile-sources.cmake")
add_library(photorec_formats OBJECT ${PHOTOREC_UPSTREAM_SOURCES})
set_target_properties(photorec_formats PROPERTIES C_VISIBILITY_PRESET hidden)
target_include_directories(photorec_formats PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PHOTOREC_ROOT}/src")
target_compile_definitions(photorec_formats PRIVATE HAVE_CONFIG_H _FILE_OFFSET_BITS=64
  fopen=die_photorec_denied_fopen rename=die_photorec_denied_rename)
target_compile_options(photorec_formats PRIVATE -O2 -ffunction-sections -fdata-sections)
add_library(diephotorec SHARED probe.cpp deny-files.c $<TARGET_OBJECTS:photorec_formats>)
set_target_properties(diephotorec PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(diephotorec PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PHOTOREC_ROOT}/src")
target_compile_features(diephotorec PRIVATE cxx_std_17)
target_compile_options(diephotorec PRIVATE -O2 -Wall -Wextra -Werror)
target_compile_definitions(diephotorec PRIVATE HAVE_CONFIG_H _FILE_OFFSET_BITS=64)
target_link_options(diephotorec PRIVATE -Wl,--no-undefined -Wl,--gc-sections)
target_link_options(diephotorec PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_property(TARGET diephotorec APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
if(ANDROID)
  target_link_options(diephotorec PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
  find_package(JNI REQUIRED)
  target_include_directories(diephotorec PRIVATE ${JNI_INCLUDE_DIRS})
endif()
